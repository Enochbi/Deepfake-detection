class EvalDataGenerator(Sequence):
    """
    Générateur simple pour l'évaluation (pas d'augmentation)
    Compatible avec l'architecture hybrid (RGB input uniquement)
    """
    def __init__(self, filepaths, labels, batch_size=64, image_size=(224, 224)):
        self.filepaths = np.array(filepaths)
        self.labels = np.array(labels, dtype=np.float32)
        self.batch_size = int(batch_size)
        self.image_size = image_size
        
        print(f"  ✅ Generator: {len(filepaths)} images, batch={batch_size}")
    
    def __len__(self):
        return max(1, int(np.ceil(len(self.filepaths) / float(self.batch_size))))
    
    def _load_image(self, path):
        """Charge l'image RGB normalisée [0, 1]"""
        try:
            with Image.open(path) as img:
                if img.mode != 'RGB':
                    img = img.convert('RGB')
                if img.size != self.image_size:
                    img = img.resize(self.image_size, Image.Resampling.BILINEAR)
                
                arr = np.array(img, dtype=np.float32)
                if np.max(arr) > 1.0:
                    arr = arr / 255.0
                
                return np.clip(arr, 0.0, 1.0)
        except Exception as e:
            print(f"⚠️  Erreur chargement {path}: {e}")
            return np.ones((*self.image_size, 3), dtype=np.float32) * 0.5
    
    def __getitem__(self, idx):
        batch_idx = np.arange(
            idx * self.batch_size,
            min((idx + 1) * self.batch_size, len(self.filepaths))
        )
        
        X = np.zeros((len(batch_idx), *self.image_size, 3), dtype=np.float32)
        y = np.zeros((len(batch_idx),), dtype=np.float32)
        
        for i, j in enumerate(batch_idx):
            X[i] = self._load_image(self.filepaths[j])
            y[i] = self.labels[j]
        
        return X, y

def load_split_from_dataset(dataset_name, split):
    """
    Charge un split spécifique d'un dataset
    Args:
        dataset_name: "DFDC", "FaceForensics++", "CelebDF_v2"
        split: "train", "val", "test"
    Returns:
        paths, labels
    """
    base = os.path.join(DATA_DIR, dataset_name, split)
    
    if not os.path.exists(base):
        print(f"⚠️  Split manquant: {dataset_name}/{split}")
        return np.array([]), np.array([])
    
    real_dir = os.path.join(base, "real")
    fake_dir = os.path.join(base, "fake")
    
    real_paths = glob.glob(os.path.join(real_dir, "*.jpg"))
    fake_paths = glob.glob(os.path.join(fake_dir, "*.jpg"))
    
    all_paths = real_paths + fake_paths
    all_labels = [0] * len(real_paths) + [1] * len(fake_paths)
    
    return np.array(all_paths), np.array(all_labels, dtype=np.int32)

# FFT LAYER (GPU)
# ==============================================================================

def fft_layer_gpu(x):
    """FFT 2D sur GPU (RGB → Gris → FFT → Log magnitude → Normalisation)"""
    gray = 0.299 * x[..., 0:1] + 0.587 * x[..., 1:2] + 0.114 * x[..., 2:3]
    gray = tf.squeeze(gray, axis=-1)
    gray_complex = tf.cast(gray, tf.complex64)
    fft_result = tf.signal.fft2d(gray_complex)
    fft_shift = tf.signal.fftshift(fft_result)
    magnitude = tf.abs(fft_shift)
    magnitude = tf.maximum(magnitude, 1e-10)
    log_magnitude = tf.math.log(1.0 + magnitude)
    min_val = tf.reduce_min(log_magnitude, axis=[1, 2], keepdims=True)
    max_val = tf.reduce_max(log_magnitude, axis=[1, 2], keepdims=True)
    normalized = tf.where(
        max_val - min_val > 1e-7,
        (log_magnitude - min_val) / (max_val - min_val + 1e-10),
        tf.zeros_like(log_magnitude)
    )
    normalized = tf.clip_by_value(normalized, 0.0, 1.0)
    return tf.stack([normalized, normalized, normalized], axis=-1)

# ==============================================================================
# GRADIENT REVERSAL LAYER (DANN)
# ==============================================================================

@tf.keras.utils.register_keras_serializable()
class GradientReversal(tf.keras.layers.Layer):
    """Gradient Reversal Layer pour DANN"""
    def __init__(self, lambda_=1.0, **kwargs):
        super().__init__(**kwargs)
        self._lambda = float(lambda_)
        self.lambda_ = None
    
    def build(self, input_shape):
        self.lambda_ = tf.Variable(
            self._lambda, 
            trainable=False, 
            dtype=tf.float32, 
            name='grl_lambda'
        )
        super().build(input_shape)
    
    def call(self, x):
        @tf.custom_gradient
        def reverse_grad(x):
            def grad(dy):
                return -self.lambda_ * dy
            return x, grad
        return reverse_grad(x)
    
    def get_config(self):
        config = super().get_config()
        config.update({"lambda_": float(self._lambda)})
        return config

def build_hybrid_model_for_eval(input_shape=(224, 224, 3)):
    """
    Reconstruit l'architecture exacte du modèle d'entraînement
    (sans les branches DANN qui ne servent pas en évaluation)
    """
    input_img = layers.Input(shape=input_shape, name="input_image")
    
    # ================= BRANCHE SPATIALE =================
    spatial_backbone = EfficientNetB0(
        include_top=False,
        weights=None,
        input_shape=input_shape,
        pooling="avg",
        name="efficientnetb0_spatial"
    )
    spatial_norm = layers.Rescaling(255.0, name="spatial_rescale")(input_img)
    F_spatial = spatial_backbone(spatial_norm)
    F_spatial = layers.BatchNormalization(name="spatial_bn")(F_spatial)
    F_spatial = layers.Dropout(0.1, name="spatial_dropout")(F_spatial)
    
    # ================= BRANCHE FRÉQUENTIELLE =================
    freq_input = layers.Lambda(fft_layer_gpu, name="fft_transform")(input_img)
    freq_backbone = EfficientNetB0(
        include_top=False,
        weights=None,
        input_shape=input_shape,
        pooling="avg",
        name="efficientnetb0_frequency"
    )
    freq_norm = layers.Rescaling(255.0, name="freq_rescale")(freq_input)
    F_freq = freq_backbone(freq_norm)
    F_freq = layers.BatchNormalization(name="freq_bn")(F_freq)
    F_freq = layers.Dropout(0.1, name="freq_dropout")(F_freq)
    
    # ================= FUSION ADAPTATIVE =================
    concat = layers.Concatenate(name="feature_concat")([F_spatial, F_freq])
    concat = layers.BatchNormalization(name="concat_bn")(concat)
    attention = layers.Dense(
        128, activation="relu",
        kernel_initializer="he_normal",
        kernel_regularizer=regularizers.l2(1e-4),
        name="attention_dense"
    )(concat)
    attention = layers.BatchNormalization(name="attention_bn")(attention)
    attention = layers.Dropout(0.2, name="attention_dropout")(attention)
    fusion_weight = layers.Dense(
        1, activation="sigmoid",
        kernel_constraint=tf.keras.constraints.MinMaxNorm(0.0, 1.0),
        name="fusion_weight"
    )(attention)
    
    def adaptive_fusion(inputs):
        w, spatial, freq = inputs
        w = tf.clip_by_value(w, 0.01, 0.99)
        return w * spatial + (1.0 - w) * freq
    
    fused = layers.Lambda(adaptive_fusion, name="adaptive_fusion")(
        [fusion_weight, F_spatial, F_freq]
    )
    fused = layers.BatchNormalization(name="fused_bn")(fused)
    
    # ================= CLASSIFICATEUR =================
    x = layers.Dense(
        256, activation="relu",
        kernel_initializer="he_normal",
        kernel_regularizer=regularizers.l2(1e-4),
        name="classifier_dense1"
    )(fused)
    x = layers.BatchNormalization(name="classifier_bn1")(x)
    x = layers.Dropout(0.3, name="classifier_dropout1")(x)
    x = layers.Dense(
        128, activation="relu",
        kernel_initializer="he_normal",
        kernel_regularizer=regularizers.l2(1e-4),
        name="classifier_dense2"
    )(x)
    x = layers.BatchNormalization(name="classifier_bn2")(x)
    output = layers.Dense(1, activation="sigmoid", name="output")(x)
    
    model = Model(inputs=input_img, outputs=output, name="hybrid_spatial_frequency")
    return model

def predict_and_evaluate(model, paths, labels, batch_size, dataset_name, use_dann=None):
    """
    Prédiction + calcul de toutes les métriques pour un dataset
    
    Args:
        model: modèle Keras
        paths: chemins des images
        labels: labels ground truth
        batch_size: taille du batch
        dataset_name: nom du dataset (pour affichage)
        use_dann: True si modèle multi-output (DANN), None pour auto-détection
    
    Returns:
        dict avec toutes les métriques
    """
    if len(paths) == 0:
        print(f"⚠️  {dataset_name}: Aucune donnée - ignoré")
        return None
    
    # Auto-détection DANN si non spécifié
    if use_dann is None:
        use_dann = len(model.outputs) > 1
    
    print(f"\n{'='*70}")
    print(f"🚀 ÉVALUATION: {dataset_name}")
    print(f"{'='*70}")
    print(f"   Échantillons: {len(paths):,}")
    print(f"   REAL: {np.sum(labels == 0):,}, FAKE: {np.sum(labels == 1):,}")
    print(f"   Mode: {'Multi-output (DANN)' if use_dann else 'Single-output'}")
    
    # Créer générateur
    eval_bs = batch_size * N_GPU
    gen = EvalDataGenerator(paths, labels, batch_size=eval_bs)
    
    # Prédiction
    print(f"\n🔮 Prédiction en cours (batch_size={eval_bs})...")
    preds = model.predict(gen, verbose=1)
    
    # Extraire probabilités
    if use_dann and isinstance(preds, (list, tuple)):
        y_pred_proba = preds[0].ravel()
    else:
        y_pred_proba = preds.ravel()
    
    # Récupérer labels
    y_true = np.concatenate([gen[i][1] for i in range(len(gen))])
    
    # Assurer même longueur
    min_len = min(len(y_true), len(y_pred_proba))
    y_true = y_true[:min_len]
    y_pred_proba = y_pred_proba[:min_len]
    
    # Prédictions binaires (seuil 0.5)
    y_pred = (y_pred_proba >= 0.5).astype(int)
    
    # ==============================================================================
    # CALCUL DES MÉTRIQUES
    # ==============================================================================
    
    metrics = {
        'dataset': dataset_name,
        'n_samples': len(y_true),
        'n_real': int(np.sum(y_true == 0)),
        'n_fake': int(np.sum(y_true == 1)),
    }
    
    # Métriques principales
    metrics['auc_roc'] = roc_auc_score(y_true, y_pred_proba)
    metrics['accuracy'] = accuracy_score(y_true, y_pred)
    metrics['f1'] = f1_score(y_true, y_pred, zero_division=0)
    metrics['precision'] = precision_score(y_true, y_pred, zero_division=0)
    metrics['recall'] = recall_score(y_true, y_pred, zero_division=0)
    
    # AUC-PR
    prec_curve, rec_curve, _ = precision_recall_curve(y_true, y_pred_proba)
    metrics['auc_pr'] = sk_auc(rec_curve, prec_curve)
    
    # Confusion matrix
    cm = confusion_matrix(y_true, y_pred)
    tn, fp, fn, tp = cm.ravel()
    
    metrics['confusion_matrix'] = cm
    metrics['tn'] = int(tn)
    metrics['fp'] = int(fp)
    metrics['fn'] = int(fn)
    metrics['tp'] = int(tp)
    metrics['specificity'] = float(tn / (tn + fp)) if (tn + fp) > 0 else 0.0
    metrics['fpr'] = float(fp / (fp + tn)) if (fp + tn) > 0 else 0.0
    
    # Courbes pour visualisation
    fpr, tpr, _ = roc_curve(y_true, y_pred_proba)
    metrics['roc_curve'] = {'fpr': fpr, 'tpr': tpr}
    metrics['pr_curve'] = {'precision': prec_curve, 'recall': rec_curve}
    
    # ==============================================================================
    # AFFICHAGE RÉSULTATS
    # ==============================================================================
    
    print(f"\n📊 RÉSULTATS {dataset_name}:")
    print(f"   🎯 AUC-ROC:      {metrics['auc_roc']:.4f}")
    print(f"   🎯 AUC-PR:       {metrics['auc_pr']:.4f}")
    print(f"   📈 Accuracy:     {metrics['accuracy']:.4f}")
    print(f"   🎲 F1-Score:     {metrics['f1']:.4f}")
    print(f"   🎪 Precision:    {metrics['precision']:.4f}")
    print(f"   📍 Recall:       {metrics['recall']:.4f}")
    print(f"   🔍 Specificity:  {metrics['specificity']:.4f}")
    
    print(f"\n   Confusion Matrix:")
    print(f"      Predicted:  REAL    FAKE")
    print(f"      REAL:      {tn:6d}  {fp:6d}")
    print(f"      FAKE:      {fn:6d}  {tp:6d}")
    
    return metrics

# ================= FONCTIONS D'AUGMENTATION FORTE =================
def random_jpeg_recompression_np(img, p=0.8):
    """JPEG recompression aléatoire - casse les artefacts de compression"""
    if np.random.rand() > p:
        return img
    # img est float32 [0,1] -> uint8
    img_u8 = (img * 255.0).astype(np.uint8)
    pil = Image.fromarray(img_u8)
    quality = np.random.randint(30, 96)
    # sauvegarde en mémoire
    buf = BytesIO()
    pil.save(buf, format='JPEG', quality=int(quality))
    buf.seek(0)
    img2 = Image.open(buf).convert('RGB')
    arr = np.array(img2, dtype=np.float32) / 255.0
    return arr

def random_gaussian_noise(img, std=0.03, p=0.7):
    """Ajout de bruit gaussien - casse le bruit caméra spécifique"""
    if np.random.rand() > p:
        return img
    noise = np.random.normal(0.0, std, img.shape).astype(np.float32)
    img = img + noise
    return np.clip(img, 0.0, 1.0)

def random_blur(img, p=0.5):
    """Flou aléatoire - casse la netteté artificielle"""
    if np.random.rand() > p:
        return img
    # Kernel size 3 or 5
    k = np.random.choice([3, 5])
    pil = Image.fromarray((img*255).astype(np.uint8))
    pil = pil.filter(ImageFilter.BoxBlur((k-1)/2))
    return np.array(pil, dtype=np.float32)/255.0

def strong_color_jitter(img, p=0.9):
    """Color jitter extrême - casse les signatures de pipeline caméra"""
    if np.random.rand() > p:
        return img
    pil = Image.fromarray((img*255).astype(np.uint8))
    # Brightness
    if np.random.rand() < 0.6:
        enhancer = ImageEnhance.Brightness(pil)
        pil = enhancer.enhance(0.6 + np.random.rand()*1.6)  # 0.6..2.2
    # Color/Saturation
    if np.random.rand() < 0.6:
        enhancer = ImageEnhance.Color(pil)
        pil = enhancer.enhance(0.6 + np.random.rand()*2.0)
    # Contrast
    if np.random.rand() < 0.4:
        enhancer = ImageEnhance.Contrast(pil)
        pil = enhancer.enhance(0.6 + np.random.rand()*1.6)
    return np.array(pil, dtype=np.float32)/255.0

def random_resize_pad(img, target=(224,224), p=0.9):
    """Random resize + pad - casse la résolution native"""
    if np.random.rand() > p:
        return img
    scale = np.random.uniform(0.7, 1.3)
    new_size = (max(1,int(scale*target[0])), max(1,int(scale*target[1])))
    pil = Image.fromarray((img*255).astype(np.uint8)).resize(new_size, Image.Resampling.BILINEAR)
    # pad/crop to target
    new = Image.new("RGB", target, (int(127),)*3)
    x = (target[0]-new_size[0])//2
    y = (target[1]-new_size[1])//2
    new.paste(pil, (max(0,x), max(0,y)))
    return np.array(new, dtype=np.float32)/255.0

class FastDataGenerator(Sequence):
    """
    Générateur OPTIMISÉ avec:
    - Augmentations fortes (training only)
    - Support domain labels pour DANN
    - Retourne seulement RGB, FFT faite dans le modèle
    """
    def __init__(self, filepaths, labels, batch_size=48, image_size=(224,224),
                 shuffle=True, augment=False, domain_labels=None):
        self.filepaths = np.array(filepaths)
        self.labels = np.array(labels, dtype=np.float32)
        self.batch_size = int(batch_size)
        self.image_size = image_size
        self.shuffle = shuffle
        self.augment = augment
        self.domain_labels = np.array(domain_labels) if domain_labels is not None else None
        self.on_epoch_end()
        # Vérification des domain labels
        if self.domain_labels is not None:
            unique_domains = np.unique(self.domain_labels)
            assert set(unique_domains).issubset({0, 1, 2}), f"Domain labels out of range: {unique_domains}"
        domain_str = f", domains={len(np.unique(domain_labels))}" if domain_labels is not None else ""
        print(f"  ✅ Generator: {len(filepaths)} images, batch={batch_size}, augment={augment}{domain_str}")

    def __len__(self):
        return max(1, int(np.ceil(len(self.filepaths) / float(self.batch_size))))

    def on_epoch_end(self):
        self.indices = np.arange(len(self.filepaths))
        if self.shuffle:
            np.random.shuffle(self.indices)

    def _load_image(self, path):
        """Charge SEULEMENT l'image RGB (pas de FFT ici!)"""
        try:
            with Image.open(path) as img:
                if img.mode != 'RGB':
                    img = img.convert('RGB')
                if img.size != self.image_size:
                    img = img.resize(self.image_size, Image.Resampling.BILINEAR)
                arr = np.array(img, dtype=np.float32)
                if np.max(arr) > 1.0:
                    arr = arr / 255.0
                arr = np.clip(arr, 0.0, 1.0)
                return arr
        except Exception:
            return np.ones((*self.image_size, 3), dtype=np.float32) * 0.5

    def _apply_strong_augmentation(self, img):
        """
        Pipeline d'augmentation forte (ordre recommandé):
        1. JPEG recompression
        2. Color jitter
        3. Gaussian noise
        4. Blur
        5. Random resize+pad
        """
        img = random_jpeg_recompression_np(img, p=0.8)
        img = strong_color_jitter(img, p=0.9)
        img = random_gaussian_noise(img, std=0.03, p=0.7)
        img = random_blur(img, p=0.5)
        img = random_resize_pad(img, target=self.image_size, p=0.9)
        return img

    def __getitem__(self, idx):
        batch_idx = self.indices[idx*self.batch_size:(idx+1)*self.batch_size]
        X = np.zeros((len(batch_idx), *self.image_size, 3), dtype=np.float32)
        y = np.zeros((len(batch_idx),), dtype=np.float32)
        for i, j in enumerate(batch_idx):
            img = self._load_image(self.filepaths[j])
            # Augmentation forte (seulement si augment=True)
            if self.augment:
                # Flip simple d'abord
                if random.random() < 0.5:
                    img = np.fliplr(img)
                # Pipeline d'augmentation forte
                img = self._apply_strong_augmentation(img)
            X[i] = img
            y[i] = self.labels[j]

        # Retour multi-output si domain labels présents
        if self.domain_labels is None:
            return X, y
        else:
            domain_batch = self.domain_labels[batch_idx]
            # Format multi-output compatible avec model.fit
            return X, {"output": y, "domain_output": domain_batch}

def extract_video_id(filepath):
    """
    Extrait le video_id depuis le chemin de fichier
    Format attendu: .../video_id_fXXX_*.jpg
    Exemples:
    - DFDC: aagfhgtpmv_f000_0.jpg → aagfhgtpmv
    - FF++: 000_003_f001_0.jpg → 000_003

    filename = os.path.basename(filepath)
    
    # Pattern général: tout avant "_fXXX_"
    match = re.match(r"(.+?)_f\d+_", filename)
    if match:
        return match.group(1)
    
    # Fallback: enlever l'extension et le dernier suffix
    return os.path.splitext(filename)[0].rsplit('_', 1)[0]

def extract_temporal_features(frame_probs):
    """
    Extrait 12 features temporelles d'une séquence de probabilités
    
    Features:
    1. mean          : Moyenne des probabilités
    2. std           : Écart-type
    3. max           : Maximum
    4. min           : Minimum
    5. range         : Étendue (max - min)
    6. skewness      : Asymétrie de la distribution
    7. kurtosis      : Aplatissement de la distribution
    8. diff_mean     : Moyenne des différences consécutives
    9. diff_std      : Écart-type des différences
    10. n_peaks_norm : Densité de pics (prob > 0.7), normalisée par longueur
    11. n_drops_norm : Densité de chutes (prob < 0.3), normalisée par longueur
    12. trend        : Pente de la tendance linéaire
    """
    if len(frame_probs) < 2:
        # Vidéo avec 1 seule frame
        return np.array([
            frame_probs[0],  # mean
            0,               # std
            frame_probs[0],  # max
            frame_probs[0],  # min
            0,               # range
            0,               # skewness
            0,               # kurtosis
            0,               # diff_mean
            0,               # diff_std
            0,               # n_peaks_norm
            0,               # n_drops_norm
            0                # trend
        ])
    
    probs = np.array(frame_probs)
    n_frames = len(probs)
    
    # Statistiques de base
    mean_prob = np.mean(probs)
    std_prob = np.std(probs)
    max_prob = np.max(probs)
    min_prob = np.min(probs)
    range_prob = max_prob - min_prob
    
    # Distribution
    skewness = scipy.stats.skew(probs)
    kurtosis = scipy.stats.kurtosis(probs)
    
    # Dynamique temporelle
    diffs = np.diff(probs)
    diff_mean = np.mean(diffs)
    diff_std = np.std(diffs)
    
    # 🔴 CORRECTION CRITIQUE: Pics et chutes
    # Pics: frames où prob > 0.7 (haute confiance FAKE)
    peaks, _ = find_peaks(probs, height=0.7)
    n_peaks = len(peaks)
    
    # Chutes: frames où prob < 0.3 (haute confiance REAL)
    # CORRECTION: Utiliser symétrie ou comptage direct
    drops, _ = find_peaks(1 - probs, height=0.7)  # Symétrique aux pics
    n_drops = len(drops)
    
    # 🟠 AMÉLIORATION: Normalisation par longueur vidéo
    # Évite le biais des vidéos longues
    n_peaks_norm = n_peaks / n_frames
    n_drops_norm = n_drops / n_frames
    
    # Tendance (régression linéaire)
    if len(probs) > 1:
        trend = np.polyfit(range(len(probs)), probs, 1)[0]
    else:
        trend = 0
    
    features = np.array([
        mean_prob, std_prob, max_prob, min_prob, range_prob,
        skewness, kurtosis,
        diff_mean, diff_std,
        n_peaks_norm, n_drops_norm,  # Maintenant normalisés
        trend
    ])
    
    # Remplacer NaN/Inf par 0
    features = np.nan_to_num(features, nan=0.0, posinf=0.0, neginf=0.0)
    
    return features

def aggregate_to_video_level(df_frames):
    """
    Agrège les frames par vidéo et extrait les features temporelles
    """
    # Grouper par vidéo
    # 🔴 CORRECTION: label avec vérification (au lieu de 'first')
    df_agg = df_frames.groupby('video_id').agg({
        'frame_prob': list,
        'label': lambda x: int(np.round(np.mean(x))),  # Majorité (robuste aux anomalies)
        'dataset': 'first'
    }).reset_index()
    
    # Extraire features temporelles
    temporal_features = []
    for probs in df_agg['frame_prob']:
        features = extract_temporal_features(probs)
        temporal_features.append(features)
    
    df_agg['temporal_features'] = temporal_features
    
    return df_agg

def evaluate_video_level(df_agg, clf, threshold, dataset_name):
    """
    Évalue un dataset au niveau vidéo
    """
    print(f"\n{'='*70}")
    print(f"🎬 ÉVALUATION VIDEO-LEVEL: {dataset_name}")
    print(f"{'='*70}")
    
    # Préparer les features
    X = np.vstack(df_agg['temporal_features'].values)
    y_true = df_agg['label'].values
    
    print(f"   Vidéos: {len(y_true):,}")
    print(f"   REAL: {np.sum(y_true == 0):,}, FAKE: {np.sum(y_true == 1):,}")
    
    # Prédire
    y_pred_proba = clf.predict_proba(X)[:, 1]
    y_pred = (y_pred_proba >= threshold).astype(int)
    
    # Calculer métriques
    metrics = {
        'dataset': dataset_name,
        'n_videos': len(y_true),
        'n_real': int(np.sum(y_true == 0)),
        'n_fake': int(np.sum(y_true == 1)),
        'auc_roc': roc_auc_score(y_true, y_pred_proba),
        'accuracy': accuracy_score(y_true, y_pred),
        'f1': f1_score(y_true, y_pred, zero_division=0),
        'precision': precision_score(y_true, y_pred, zero_division=0),
        'recall': recall_score(y_true, y_pred, zero_division=0),
    }
    
    # 🟠 AMÉLIORATION: AUC-PR avec average_precision_score (plus standard)
    from sklearn.metrics import average_precision_score
    metrics['auc_pr'] = average_precision_score(y_true, y_pred_proba)
    
    # 🔴 CORRECTION: Confusion matrix avec garde
    cm = confusion_matrix(y_true, y_pred)
    
    if cm.shape == (2, 2):
        tn, fp, fn, tp = cm.ravel()
    else:
        # Dataset déséquilibré avec une seule classe prédite
        print(f"   ⚠️  Confusion matrix non standard: {cm.shape}")
        if cm.shape == (1, 1):
            # Une seule classe (pathologique)
            if y_true[0] == 0:  # Tout REAL
                tn, fp, fn, tp = cm[0, 0], 0, 0, 0
            else:  # Tout FAKE
                tn, fp, fn, tp = 0, 0, 0, cm[0, 0]
        elif cm.shape == (2, 1):
            # 2 classes vraies, 1 classe prédite
            tn = cm[0, 0] if y_pred[0] == 0 else 0
            fp = cm[0, 0] if y_pred[0] == 1 else 0
            fn = cm[1, 0] if y_pred[0] == 0 else 0
            tp = cm[1, 0] if y_pred[0] == 1 else 0
        else:
            # Fallback
            tn = fp = fn = tp = 0
    
    metrics['confusion_matrix'] = cm
    metrics['tn'] = int(tn)
    metrics['fp'] = int(fp)
    metrics['fn'] = int(fn)
    metrics['tp'] = int(tp)
    metrics['specificity'] = float(tn / (tn + fp)) if (tn + fp) > 0 else 0.0
    metrics['fpr'] = float(fp / (fp + tn)) if (fp + tn) > 0 else 0.0
    
    # Courbes
    fpr, tpr, _ = roc_curve(y_true, y_pred_proba)
    metrics['roc_curve'] = {'fpr': fpr, 'tpr': tpr}
    
    prec_curve, rec_curve, _ = precision_recall_curve(y_true, y_pred_proba)
    metrics['pr_curve'] = {'precision': prec_curve, 'recall': rec_curve}
    
    # Afficher
    print(f"\n📊 RÉSULTATS:")
    print(f"   🎯 AUC-ROC:      {metrics['auc_roc']:.4f}")
    print(f"   🎯 AUC-PR:       {metrics['auc_pr']:.4f}")
    print(f"   📈 Accuracy:     {metrics['accuracy']:.4f}")
    print(f"   🎲 F1-Score:     {metrics['f1']:.4f}")
    print(f"   🎪 Precision:    {metrics['precision']:.4f}")
    print(f"   📍 Recall:       {metrics['recall']:.4f}")
    print(f"   🔍 Specificity:  {metrics['specificity']:.4f}")
    
    print(f"\n   Confusion Matrix:")
    print(f"      Predicted:  REAL    FAKE")
    print(f"      REAL:      {tn:6d}  {fp:6d}")
    print(f"      FAKE:      {fn:6d}  {tp:6d}")
    
    return metrics
